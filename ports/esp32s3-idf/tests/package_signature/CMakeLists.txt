cmake_minimum_required(VERSION 3.21)
project(jfsg_signature_qualification LANGUAGES C CXX)
include(CTest)
set(JFSG_MBEDTLS_SOURCE_DIR "" CACHE PATH "Exact ESP-IDF components/mbedtls/mbedtls source tree")
if(NOT EXISTS "${JFSG_MBEDTLS_SOURCE_DIR}/library/ecdsa.c")
  message(FATAL_ERROR "Set JFSG_MBEDTLS_SOURCE_DIR to the IDF mbedTLS source tree")
endif()
get_filename_component(runtime_root "${CMAKE_CURRENT_SOURCE_DIR}/../../../.." ABSOLUTE)
set(crypto_sources bignum.c bignum_core.c bignum_mod.c bignum_mod_raw.c
    constant_time.c ecp.c ecp_curves.c ecp_curves_new.c ecdsa.c
    asn1parse.c asn1write.c sha256.c platform_util.c)
list(TRANSFORM crypto_sources PREPEND "${JFSG_MBEDTLS_SOURCE_DIR}/library/")
add_library(jfsg_mbedcrypto STATIC ${crypto_sources})
target_include_directories(jfsg_mbedcrypto PUBLIC "${JFSG_MBEDTLS_SOURCE_DIR}/include" "${CMAKE_CURRENT_SOURCE_DIR}")
target_compile_definitions(jfsg_mbedcrypto PUBLIC MBEDTLS_CONFIG_FILE="mbedtls_test_config.h")
set(adapter "${runtime_root}/ports/esp32s3-idf/main/jellyframe_esp32s3_package_signature.cpp")
add_executable(jfsg_signature_tests signature_tests.cpp "${adapter}"
    "${runtime_root}/src/device_runtime_contracts/device_signed_package.cpp")
target_compile_features(jfsg_signature_tests PRIVATE cxx_std_17)
target_compile_definitions(jfsg_signature_tests PRIVATE JELLYFRAME_PACKAGE_SIGNATURE_TESTING=1)
target_include_directories(jfsg_signature_tests PRIVATE "${runtime_root}/src" "${runtime_root}/ports/esp32s3-idf/main")
target_link_libraries(jfsg_signature_tests PRIVATE jfsg_mbedcrypto)
# Compile the production translation unit separately without test injection.
add_library(jfsg_signature_production OBJECT "${adapter}")
target_compile_features(jfsg_signature_production PRIVATE cxx_std_17)
target_include_directories(jfsg_signature_production PRIVATE "${runtime_root}/src")
target_link_libraries(jfsg_signature_production PRIVATE jfsg_mbedcrypto)
foreach(case valid tamper wrong_key invalid_scalars arguments parser)
  add_test(NAME jfsg_crypto_${case} COMMAND jfsg_signature_tests ${case})
endforeach()
